cmake_minimum_required(VERSION 3.20)
project(RetroNP VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options("$<$<CONFIG:Release>:/O2>")
else()
  add_compile_options("$<$<CONFIG:Release>:-O2>")
endif()
add_executable(RetroNP WIN32 src/main.cpp)
target_compile_definitions(RetroNP PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(RetroNP PRIVATE gdi32 user32 shell32)
if(MINGW)
  target_link_options(RetroNP PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(RetroNP PRIVATE /W4 /utf-8)
else()
  target_compile_options(RetroNP PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(editor_tests tests/editor_tests.cpp)
target_compile_definitions(editor_tests PRIVATE NOMINMAX)
enable_testing()
add_test(NAME editor_tests COMMAND editor_tests)
add_executable(io_tests tests/io_tests.cpp)
target_compile_definitions(io_tests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(io_tests PRIVATE gdi32 user32 shell32)
if(MINGW)
  target_link_options(io_tests PRIVATE -static -static-libgcc -static-libstdc++)
endif()
add_test(NAME io_tests COMMAND io_tests)
